cmake_minimum_required(VERSION 3.24)
project(context_overlay LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
option(CONTEXT_OVERLAY_BUILD_TESTS "Build unit tests" ON)
if(WIN32)
  add_executable(context_overlay WIN32
    src/main.cpp src/overlay_window.cpp src/overlay_interaction.cpp
    src/dwell_coordinator.cpp src/uia_identity.cpp)
  target_compile_definitions(context_overlay PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
  target_link_libraries(context_overlay PRIVATE user32 gdi32 ole32 oleaut32 advapi32)
  if(MSVC)
    target_compile_options(context_overlay PRIVATE /W4 /permissive- /utf-8)
  elseif(MINGW)
    target_compile_options(context_overlay PRIVATE -Wall -Wextra -Wpedantic -municode)
    target_link_options(context_overlay PRIVATE -municode)
  endif()
endif()
if(CONTEXT_OVERLAY_BUILD_TESTS)
  enable_testing()
  add_executable(identity_hash_test tests/identity_hash_test.cpp)
  target_compile_features(identity_hash_test PRIVATE cxx_std_20)
  add_test(NAME identity_hash COMMAND identity_hash_test)
  add_executable(theme_tokens_test tests/theme_tokens_test.cpp)
  target_compile_features(theme_tokens_test PRIVATE cxx_std_20)
  add_test(NAME theme_tokens COMMAND theme_tokens_test)
  add_executable(style_profile_test tests/style_profile_test.cpp)
  target_compile_features(style_profile_test PRIVATE cxx_std_20)
  add_test(NAME style_profile COMMAND style_profile_test)
  add_executable(renderer_alpha_test tests/renderer_alpha_test.cpp)
  target_compile_features(renderer_alpha_test PRIVATE cxx_std_20)
  add_test(NAME renderer_alpha COMMAND renderer_alpha_test)
  add_executable(hotkey_state_test tests/hotkey_state_test.cpp)
  target_compile_features(hotkey_state_test PRIVATE cxx_std_20)
  add_test(NAME hotkey_state COMMAND hotkey_state_test)
  if(WIN32)
    add_executable(hotkey_registration_test tests/test_hotkey_registration.cpp src/overlay_interaction.cpp)
    target_compile_definitions(hotkey_registration_test PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(hotkey_registration_test PRIVATE user32)
    add_test(NAME hotkey_registration COMMAND hotkey_registration_test)
  endif()
endif()
